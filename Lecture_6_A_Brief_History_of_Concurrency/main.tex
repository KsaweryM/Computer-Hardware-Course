}
\end{lstlisting}
  \end{minipage}
  \codenote[-0.6cm]{Keep the ISR in fast memory}{The program normally lives in the \textbf{flash}: big, slow, and sometimes busy. An ISR must start \textbf{at once}, so \texttt{IRAM\_ATTR} puts it into \textbf{IRAM}, fast memory inside the chip.}

  \keymsg{Every ISR on the ESP32 gets \texttt{IRAM\_ATTR}.}

\end{lstlisting}

\begin{frame}{The stack}
  \begin{columns}[c,onlytextwidth]
    \column{0.44\textwidth}
      \centering
      \resizebox{0.9\linewidth}{!}{\stackonly}
    \column{0.52\textwidth}
      \defbox{The \textbf{stack} is a part of the memory where a program keeps values \textbf{for later},
        for example the local variables of a function.}
      \stepbox{The top}{The register \texttt{sp} (\emph{stack pointer}) holds the address of the top.
        From the top down: \textbf{in use}. Above it: free.}

\begin{frame}[fragile]{A local variable lives on the stack}
  \begin{columns}[T,onlytextwidth]
    \column{0.4\textwidth}
      {\scriptsize\textbf{C}\par}
\begin{lstlisting}[style=rtos]
int f(void) {
    int a = 7;
    return a + 1;
}
\end{lstlisting}
      \stepbox{One local variable}{\texttt{a} exists only while \texttt{f} runs. Where does it live?}
    \column{0.56\textwidth}
      {\scriptsize\textbf{RISC-V} (gcc \texttt{-O0 -fomit-frame-pointer})\par}
\begin{lstlisting}[style=rv6]
f:
  addi sp, sp, -16   # make room
  li   a5, 7
  sw   a5, 12(sp)    # a = 7, on the stack
  lw   a5, 12(sp)    # read a
  addi a5, a5, 1
  mv   a0, a5        # the return value
  addi sp, sp, 16    # free the room
  jr   ra            # return
\end{lstlisting}
  \end{columns}
  \vfill
  \keymsg{The function takes room on the stack at the start, and gives it back at the end.}

\begin{frame}{A local variable lives on the stack: step by step}

    the result (8) in \texttt{a0}.}}%

\begin{frame}{Inside the CPU: 2. the prologue}
  \begin{center}

  \end{center}

  \only<2>{\stepbox{\texttt{sw t0, 12(sp)}}{Save the main program's \texttt{t0} (1): the body is about to use \texttt{t0}.}}%
  \only<3>{\stepbox{\texttt{sw t1, 8(sp)}}{Save the main program's \texttt{t1} (5): the body uses \texttt{t1} too.}}

\begin{frame}{Inside the CPU: 3. the body}
  \begin{center}

  \only<1>{\stepbox{\texttt{la t0, presses}}{\texttt{t0} = the \textbf{address} of \texttt{presses}: 0x1004. The ISR cannot know what the main program keeps in its registers, so it gets the address itself. (\texttt{la} is two real instructions, 8 bytes: the next one is at 0x0214.)}}%
  \only<2>{\stepbox{\texttt{lw t1, 0(t0)}}{\texttt{t1} = \texttt{presses} = 2.}}%
  \only<3>{\stepbox{\texttt{addi t1, t1, 1}}{\texttt{t1} = 3.}}%
  \only<4>{\stepbox{\texttt{sw t1, 0(t0)}}{\texttt{presses} = 3. These four instructions are \texttt{presses++}: the work of the ISR.}}
\end{frame}

\begin{frame}{Inside the CPU: 4. the epilogue and return}
  \begin{center}
    \resizebox{!}{0.63\textheight}{\only<1>{\irqpass{12}}\only<2>{\irqpass{13}}\only<3>{\irqpass{14}}\only<4>{\irqpass{15}}}
  \end{center}
  \vfill
  \only<1>{\stepbox{\texttt{lw t1, 8(sp)}}{Restore the main program's \texttt{t1}: 5.}}%
  \only<2>{\stepbox{\texttt{lw t0, 12(sp)}}{Restore the main program's \texttt{t0}: 1.}}%
  \only<3>{\stepbox{\texttt{addi sp, sp, 16}}{Free the room: \texttt{sp} is 0x2000 again. The stack is as before the interrupt.}}%
  \only<4>{\stepbox{\texttt{mret}}{PC $\leftarrow$ \texttt{mepc} = 0x0108. The main program continues with \texttt{sw}, with \texttt{t0} = 1 and \texttt{t1} = 5: it \textbf{does not know} that it was interrupted.}}
\end{frame}

\begin{frame}{The rule: save what you change}
  \vfill
  \defbox{The hardware saves \textbf{only the PC} (in \texttt{mepc}). Every register that the ISR changes, the ISR must
    \textbf{save} in its prologue and \textbf{restore} in its epilogue.}
  \vspace{0.2cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \stepbox{Our ISR}{Its body changes \texttt{t0} and \texttt{t1}. So the prologue saves exactly these two: that is enough.}
    \column{0.48\textwidth}
      \stepbox{\texttt{sp}?}{It changes too, but the epilogue moves it back by the same 16 bytes.}
  \end{columns}
  \vfill
  \keymsg{Then the main program finds all its registers exactly as it left them.}
\end{frame}

\begin{frame}[fragile]{But our ISR is written in C}

\begin{lstlisting}[style=rtos]

\end{lstlisting}

    \column{0.56\textwidth}

\begin{lstlisting}[style=rv6]

\begin{frame}{The problem: nobody saves \texttt{a5}}
  \vfill
  \begin{center}
    \ctrouble
  \end{center}
  \vfill
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \stepbox{The rule of C}{A function \textbf{may} change \texttt{t0} to \texttt{t6} and \texttt{a0} to \texttt{a7} without saving them.
        If the caller still needs one of them, the \textbf{caller} saves it before the call.}
    \column{0.48\textwidth}
      \stepbox{But nobody called the ISR}{The main program was \textbf{interrupted}, between any two instructions.
        It had no chance to save \texttt{a5}. And the C function does not save it either.}
  \end{columns}
  \vfill
  \keymsg{So somebody must save the registers \textbf{before} our C function runs, and restore them after it.}
\end{frame}

\begin{frame}[fragile]{The interrupt entry code: the real one}
  \begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
\begin{lstlisting}[style=rv6]
_interrupt_handler:
  addi sp, sp, -128   # room for 32 registers
  sw   ra, 4(sp)
  sw   t0, 20(sp)
  sw   a5, 60(sp)
  ...                 # 29 sw in total
  csrr t0, mepc       # mepc too
  sw   t0, 0(sp)
  call rtos_int_enter
  jal  _global_interrupt_handler
  call rtos_int_exit
  lw   ra, 4(sp)
  ...                 # 29 lw in total
  addi sp, sp, 128
  mret
\end{lstlisting}
    \column{0.46\textwidth}
      \stepbox{Is the picture simplified?}{Not much. This is the real code of the ESP32-C3 libraries (ESP-IDF),
        shortened only at the dots.}
      \stepbox{Prologue and epilogue}{All registers and \texttt{mepc} go to the stack. At the end: all back, and \texttt{mret}.}
      \stepbox{The body}{\texttt{\_global\_interrupt\_handler} looks into the table and calls our \texttt{onButtonA}
        with a normal call. \texttt{rtos\_int\_enter} and \texttt{exit} tell FreeRTOS that an interrupt runs.}
  \end{columns}
  \vfill
  \keymsg{Remember: \textbf{all} registers of the interrupted program are on its stack. The scheduler needs this.}
\end{frame}

\begin{frame}[fragile]{Can an ISR be faster?}
  \begin{columns}[T,onlytextwidth]
    \column{0.44\textwidth}
\begin{lstlisting}[style=rtos]
void __attribute__((interrupt))
onButtonA() {
  presses++;
}
\end{lstlisting}
      {\scriptsize\textbf{The compiler makes} (gcc \texttt{-O2})\par}
\begin{lstlisting}[style=rv6]
  addi sp, sp, -16
  sw   a4, 12(sp)    # only a4
  sw   a5, 8(sp)     # and a5
  ...                # presses++
  lw   a4, 12(sp)
  lw   a5, 8(sp)
  addi sp, sp, 16
  mret
\end{lstlisting}
    \column{0.52\textwidth}
      \stepbox{Save only what you use}{With \texttt{interrupt}, the compiler writes the prologue and the epilogue itself,
        exactly like our ISR in assembly: 2 registers instead of 29, and \texttt{mret} at the end.
        An ISR written by hand in assembly can do the same.}
      \stepbox{When is it worth it?}{When interrupts come \textbf{very often}, for example 100\,000 times per second
        to control a motor or to read sound samples. Then every saved instruction counts.}
      \probbox{The price}{Such an ISR skips the system: it cannot use FreeRTOS functions or wake a task, and it works on one
        chip only. The Arduino core does not even offer it: there the entry code always runs first.}
  \end{columns}
\end{frame}

\begin{frame}{A timer is a stopwatch with an alarm}
  \vfill
  \begin{center}
    \timerblock
  \end{center}
  \vfill
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \stepbox{It counts by itself}{A \textbf{timer} is a counter in hardware. A clock gives it pulses at a fixed rate,
        and at every pulse it adds 1. The CPU does not have to do anything.}
    \column{0.48\textwidth}
      \stepbox{It rings}{When the counter reaches the set value (the \textbf{alarm}), the timer raises an
        \textbf{interrupt}. Then it can start again from 0, and ring again.}
  \end{columns}
  \vfill
\end{frame}

\begin{frame}{Interrupts from a timer}
  \vfill
  \begin{center}
    \timersaw
  \end{center}
  \vfill
  \keymsg{A timer interrupt lets the processor do something \textbf{regularly}, without checking the time in the main program.}
\end{frame}

\begin{frame}{Where is the timer?}
  \begin{center}
    \resizebox{0.82\textwidth}{!}{\timerwhere}
  \end{center}
  \vspace{-0.1cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
      \stepbox{\badge{1}\ In the core}{Many processors have a timer in the core itself: every core of a PC has one,
        and RISC-V defines one (\texttt{mtime}). The operating system takes its regular interrupt from it.}
    \column{0.32\textwidth}
      \stepbox{\badge{2}\ Next to the core}{A device on the same chip, like the GPIO controller. The ESP32-C3 has
        \textbf{two} such timers for us, and one more for the system.}
    \column{0.32\textwidth}
      \stepbox{\badge{3}\ On another chip}{For example a real-time clock with an alarm. Its signal comes on a wire to a
        GPIO pin: for the CPU it is a GPIO interrupt, just like button A.}
  \end{columns}
\end{frame}

\begin{frame}{Three LEDs, three rhythms, one timer}

  \defbox{One timer interrupt at a fixed rate is called a \textbf{tick}. Here: a tick every 100 ms.
    At every tick, the ISR \textbf{decides} what to do: LED 1 every 3 ticks, LED 2 every 5, LED 3 every 7.}

\begin{frame}[fragile]{The tick in the code}
  \begin{columns}[c,onlytextwidth]
    \column{0.62\textwidth}
\begin{lstlisting}[style=rtos]
volatile unsigned long ticks = 0;
volatile bool led1 = false, led2 = false, led3 = false;

void IRAM_ATTR onTick() {        // every 100 ms
  ticks++;
  if (ticks % 3 == 0) { led1 = !led1; digitalWrite(LED1, led1); }
  if (ticks % 5 == 0) { led2 = !led2; digitalWrite(LED2, led2); }
  if (ticks % 7 == 0) { led3 = !led3; digitalWrite(LED3, led3); }
}
\end{lstlisting}
    \column{0.35\textwidth}
      \stepbox{\texttt{ticks \% 3 == 0}}{Every third tick: 300 ms.}
      \stepbox{One timer}{\texttt{timerAlarm(timer, 100000, true, 0)}: a tick every 100 ms.}
  \end{columns}
  \vfill
  \keymsg{Remember this idea: a \textbf{regular tick}, and a \textbf{decision} at every tick. We will need it in Part 2.}
\end{frame}

\begin{frame}{Context switch, step by step}
  \begin{center}

      \only<1>{\ctxswitch{1}}%
      \only<2>{\ctxswitch{2}}%
      \only<3>{\ctxswitch{3}}%

  \vspace{-0.15cm}
  \only<1>{\stepbox{\badge{1}\ Tick}{Task A runs. The timer interrupt (the tick) arrives: \texttt{mepc} $\leftarrow$ PC, and the CPU jumps to the interrupt entry code.}}
  \only<2>{\stepbox{\badge{2}\ Save A}{The entry code puts A's registers and its PC (from \texttt{mepc}) onto \textbf{stack A}, as we saw.

  \only<3>{\stepbox{\badge{3}\ Choose B}{The scheduler picks the next task: \textbf{B}.}}

\begin{frame}[fragile]{Saving and restoring registers (simplified)}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      {\scriptsize\textbf{Save} the old task (step 2)\par}
\begin{lstlisting}[style=rv6]
addi sp, sp, -CONTEXT_SIZE # room on its stack
sw   ra, 1*4(sp)           # save registers
sw   t0, 2*4(sp)
...
csrr t0, mepc              # where it stopped
sw   t0, 0(sp)             # save the PC
lw   t0, pxCurrentTCB      # its TCB
sw   sp, 0(t0)             # pxTopOfStack = sp
\end{lstlisting}
    \column{0.48\textwidth}
      {\scriptsize\textbf{Restore} the new task (step 4)\par}
\begin{lstlisting}[style=rv6]
lw   t0, pxCurrentTCB      # the new TCB
lw   sp, 0(t0)             # sp = pxTopOfStack
lw   t0, 0(sp)             # its saved PC
csrw mepc, t0
lw   ra, 1*4(sp)           # restore registers
lw   t0, 2*4(sp)
...
addi sp, sp, CONTEXT_SIZE
mret                       # continue the task
\end{lstlisting}
  \end{columns}
  \vfill
  \keymsg{Exactly our 4 steps, in code: \texttt{sw} and \texttt{lw} for the registers, and one TCB per task.}
\end{frame}

\begin{frame}[fragile]{The context switch on the ESP32-C3: the real code}
  \begin{columns}[T,onlytextwidth]
    \column{0.6\textwidth}
\begin{lstlisting}[style=rv6, escapeinside={(*}{*)}]
_interrupt_handler:          # the tick arrives
  addi sp, sp, -128
  sw   ...                   # (*\badge{2}*) all registers of A
  call rtos_int_enter        #   on stack A
     lw  a0, pxCurrentTCB
     sw  sp, 0(a0)           #   A's top = sp
  jal  _global_interrupt_handler
  call rtos_int_exit
     call vTaskSwitchContext # (*\badge{3}*) choose B
     lw  a0, pxCurrentTCB
     lw  sp, 0(a0)           # (*\badge{4}*) sp = B's top
  lw   ...                   #   all registers of B
  addi sp, sp, 128           #   from stack B
  mret                       # B continues
\end{lstlisting}
    \column{0.37\textwidth}
      \stepbox{The same entry code}{It is \texttt{\_interrupt\_handler} from Part 1, with the inside of
        \texttt{rtos\_int\_enter} and \texttt{rtos\_int\_exit} shown below them (indented).}
      \stepbox{The trick}{The entry code \textbf{always} saves all registers on the stack of the task that was interrupted.
        So to switch, it is enough to change \texttt{sp} before the epilogue: the epilogue then restores \textbf{B}.}
  \end{columns}
  \vfill
  \keymsg{A context switch is an interrupt that returns to a \textbf{different} task.}
\end{frame}

\begin{frame}[fragile]{Solution: a race between two tasks}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
\begin{lstlisting}[style=rtos]
void setup() {
  // the LCD as before
  xTaskCreate(incTask, "incA", 2048,
              (void *)&doneA, 1, NULL);
  xTaskCreate(incTask, "incB", 2048,
              (void *)&doneB, 1, NULL);
}

void loop() {
  if (doneA && doneB) {
    lcd.setCursor(0, 0);
    lcd.print("Goal: ");  lcd.print(2 * N);
    lcd.setCursor(0, 1);
    lcd.print("Got:  ");  lcd.print(counter);
    while (true) delay(1000);
  }
  delay(10);            // wait without wasting the CPU
}
\end{lstlisting}
    \column{0.42\textwidth}
      \stepbox{Question 3}{\texttt{N} is \textbf{smaller} than 2\,000\,000 and \textbf{different} every time.
        It is never correct.}
      \stepbox{If you finished early}{With \texttt{N = 1000} each task finishes within one time slice:
        no switch comes in the middle, so the result is correct.}
  \end{columns}

\begin{frame}{What happened?}
  \begin{center}
    \racetimeline
  \end{center}
  \vspace{-0.2cm}
  \defbox{This is a \textbf{race condition}: the result depends on \textbf{when} the scheduler switches the tasks.}
  \stepbox{We saw it three times}{Two cores: two presses, \texttt{presses} grew by one.
    \quad Simulator 6: 6 additions, counter 4 or 5.
    \quad Task 5: two million additions, the counter got less.}
  \vfill
  \begin{alertblock}{The question}
    \small Why? \texttt{counter++} is \textbf{one line}. How can a switch fall \emph{in the middle} of it?
  \end{alertblock}
\end{frame}
